Messy CSV cleaner

Removes duplicate rows, trims spaces, fixes name casing, lowercases emails and strips phone formatting.

Input

Nome, Email ,Telefone

JOAO LIMA,joao@exemplo.com,41 98888-2222
Ana Souza,ana@exemplo.com,+55 41 99999-1111
maria  dias,,41977773333
carlos nunes,Carlos@exemplo.com , 41-96666-4444

Output

nome,email,telefone
Ana Souza,ana@exemplo.com,41999991111
Joao Lima,joao@exemplo.com,41988882222
Carlos Nunes,carlos@exemplo.com,41966664444
Python source
"""Limpa um CSV: remove duplicados, apara espacos, padroniza e-mails e telefones."""
import csv, re, sys

def limpar(entrada, saida):
    vistos, linhas = set(), []
    with open(entrada, newline="", encoding="utf-8") as f:
        for r in csv.DictReader(f):
            r = {k.strip().lower(): (v or "").strip() for k, v in r.items()}
            r["nome"] = r["nome"].title()
            r["email"] = r["email"].lower()
            r["telefone"] = re.sub(r"\D", "", r["telefone"])
            if r["email"] in vistos or not r["email"]:
                continue
            vistos.add(r["email"])
            linhas.append(r)
    with open(saida, "w", newline="", encoding="utf-8") as f:
        w = csv.DictWriter(f, fieldnames=["nome", "email", "telefone"])
        w.writeheader(); w.writerows(linhas)
    return len(linhas)

if __name__ == "__main__":
    print(limpar(sys.argv[1], sys.argv[2]), "linhas salvas")

Run: python limpar_csv.py contatos_sujos.csv limpo.csv

"""Le um CSV de vendas e gera resumo por produto e por mes (total, quantidade, ticket medio)."""
import csv, sys
from collections import defaultdict

def resumir(arquivo):
    prod, mes = defaultdict(lambda: [0, 0.0]), defaultdict(float)
    with open(arquivo, newline="", encoding="utf-8") as f:
        for r in csv.DictReader(f):
            v = float(r["valor"].replace(",", "."))
            prod[r["produto"]][0] += int(r["qtd"]); prod[r["produto"]][1] += v
            mes[r["data"][:7]] += v
    print("POR PRODUTO"); 
    for p, (q, t) in sorted(prod.items(), key=lambda x: -x[1][1]):
        print(f"  {p:<10} qtd={q:<3} total=R$ {t:>8.2f} ticket=R$ {t/q:>7.2f}")
    print("POR MES")
    for m, t in sorted(mes.items()): print(f"  {m}  R$ {t:>8.2f}")

if __name__ == "__main__": resumir(sys.argv[1])

Run: python resumo_vendas.py vendas.csv

Page heading extractor

Reads a web page (local file or URL) and saves every h1, h2 and h3 to a CSV.

Input

<html><body><h1>Loja Exemplo</h1><h2>Camisetas</h2><h3>Camiseta Basica</h3><h2>Canecas</h2><h3>Caneca 300ml</h3></body></html>

Output

nivel,texto
h1,Loja Exemplo
h2,Camisetas
h3,Camiseta Basica
h2,Canecas
h3,Caneca 300ml
Python source
"""Baixa uma pagina HTML e salva titulos (h1-h3) e links em CSV."""
import csv, sys, urllib.request
from html.parser import HTMLParser

class P(HTMLParser):
    def __init__(s): super().__init__(); s.rows=[]; s.tag=None; s.href=None; s.buf=""
    def handle_starttag(s, t, a):
        if t in ("h1","h2","h3"): s.tag=t; s.buf=""
        if t=="a": s.href=dict(a).get("href")
    def handle_data(s, d):
        if s.tag: s.buf+=d
    def handle_endtag(s, t):
        if t==s.tag and s.buf.strip(): s.rows.append((t, s.buf.strip(), "")); s.tag=None

def extrair(origem, saida):
    html = open(origem, encoding="utf-8").read() if not origem.startswith("http") else urllib.request.urlopen(origem, timeout=20).read().decode("utf-8","ignore")
    p = P(); p.feed(html)
    with open(saida,"w",newline="",encoding="utf-8") as f:
        w=csv.writer(f); w.writerow(["nivel","texto"]); w.writerows([(a,b) for a,b,_ in p.rows])
    return len(p.rows)

if __name__ == "__main__": print(extrair(sys.argv[1], sys.argv[2]), "titulos salvos")

Run: python extrair_titulos.py pagina.html titulos.csv
